Service Area & Gap Script
daycare_service_areas.py
PyQGIS • Processing 
# -------------------------------------------------------------------------
# DAYCARE ACCESSIBILITY: WALKING SERVICE AREAS (PyQGIS)
# Network: OpenStreetMap pedestrian ways | Speed: 4.8 km/h
# -------------------------------------------------------------------------
import processing
from qgis.core import QgsProject

WALK_SPEED_KMH = 4.8
CUTOFFS_MIN = [5, 10, 15]

def service_areas(network, facilities):
    results = {}
    for minutes in CUTOFFS_MIN:
        seconds = minutes * 60
        results[minutes] = processing.run("native:serviceareafromlayer", {
            "INPUT": network,
            "STRATEGY": 1,                      # shortest travel time
            "DEFAULT_SPEED": WALK_SPEED_KMH,
            "START_POINTS": facilities,
            "TRAVEL_COST2": seconds,
            "OUTPUT_LINES": "memory:",
        })["OUTPUT_LINES"]
    return results

def find_gaps(residential, catchment_15):
    # Residential blocks that fall outside the 15-minute catchment
    return processing.run("native:difference", {
        "INPUT": residential,
        "OVERLAY": catchment_15,
        "OUTPUT": "memory:",
    })["OUTPUT"]
