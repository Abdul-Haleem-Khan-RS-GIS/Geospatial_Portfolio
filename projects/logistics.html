multimodal_commute_solver.py

# -------------------------------------------------------------------------
# ENTERPRISE WORKFORCE MULTIMODAL ROUTING & ISOCHRONE SOLVER
# CRS: EPSG:2056 (CH1903+ / LV95) | Impedance: DEM Gradient + Turn Cost
# -------------------------------------------------------------------------

import osmnx as ox
import networkx as nx
import geopandas as gpd
import numpy as np

def build_multimodal_impedance_network(place_query: "Zurich, Switzerland", dem_raster_path: str):
    # 1. Ingest walkable, bikeable, and road vectors via OSM Overpass API
    G = ox.graph_from_place(place_query, network_type="all", simplify=True)
    G = ox.project_graph(G, to_crs="EPSG:2056")
    
    # 2. Attach swissALTI3D topographic elevation & compute edge slopes
    G = ox.elevation.add_node_elevations_raster(G, dem_raster_path)
    G = ox.elevation.add_edge_grades(G)
    
    # 3. Calculate multimodal cost weight (Tobler Hiking Function modifier)
    for u, v, k, data in G.edges(keys=True, data=True):
        grade = data.get('grade', 0.0)
        length_m = data.get('length', 1.0)
        mode = data.get('highway', 'residential')
        
        # Exponential resistance for steep uphill cycling gradients (> 6%)
        slope_penalty = np.exp(3.5 * abs(grade)) if grade > 0.06 else 1.0
        base_speed = 18.0 if mode == 'cycleway' else 12.0 # km/h
        data['impedance_time_sec'] = (length_m / (base_speed * 0.27778)) * slope_penalty

    return G

# Compute Origin-Destination Cost Matrix across 1,420 commuter locations
def generate_hq_commute_isochrones(G, hq_node_id, intervals=[15, 30, 45]):
    sheds = {}
    for trip_time in intervals:
        subgraph = nx.ego_graph(G, hq_node_id, radius=trip_time * 60, distance='impedance_time_sec')
        sheds[trip_time] = gpd.GeoSeries([gpd.points_from_xy([data['x'] for _, data in subgraph.nodes(data=True)],
                                                             [data['y'] for _, data in subgraph.nodes(data=True)]).unary_union.convex_hull],
                                         crs="EPSG:2056")
    return sheds
