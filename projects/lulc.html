// Google Earth Engine Sentinel-2 Random Forest LULC Pipeline
// Author: Abdul Haleem Khan | GIS Analyst & Remote Sensing Specialist

var aoi = ee.Geometry.Polygon([...]); // Regional Watershed Boundary
var s2 = ee.ImageCollection('COPERNICUS/S2_SR_HARMONIZED')
  .filterBounds(aoi)
  .filterDate('2024-05-01', '2024-09-30')
  .filter(ee.Filter.lt('CLOUDY_PIXEL_PERCENTAGE', 15))
  .map(function(img) {
    var qa = img.select('QA60');
    var mask = qa.bitwiseAnd(1 << 10).or(qa.bitwiseAnd(1 << 11)).not();
    return img.updateMask(mask).divide(10000);
  });

// Extract Median Surface Composite & Spectral Indices
var composite = s2.median().clip(aoi);
var ndvi = composite.normalizedDifference(['B8', 'B4']).rename('NDVI');
var ndwi = composite.normalizedDifference(['B3', 'B8']).rename('NDWI');
var ndbi = composite.normalizedDifference(['B11', 'B8']).rename('NDBI');
var featureStack = composite.select(['B2','B3','B4','B8','B11','B12'])
  .addBands([ndvi, ndwi, ndbi]);

// Train Supervised Random Forest Classifier (150 Trees)
var training = featureStack.sampleRegions({
  collection: trainingPolygons,
  properties: ['class_code'],
  scale: 10
});
var classifier = ee.Classifier.smileRandomForest(150).train(training, 'class_code');
var classified = featureStack.classify(classifier);

// Validation against 30% Test Dataset
var testAccuracy = classified.errorMatrix('class_code', 'classification');
print('Overall Kappa:', testAccuracy.kappa());
print('Accuracy Matrix:', testAccuracy.accuracy());
